# 11. quiz.py

class Quiz:
    """농구 퀴즈를 표현하는 클래스"""
    
    def __init__(self, question, choices, answer, hint=""):
        """
        question (str): 퀴즈 문제
        choices (list): 4개의 선택지
        answer (int): 정답 번호 (1~4, JSON에서는 0~3 인덱스)
        hint (str): 힌트 (선택사항)
        """
        self.question = question
        self.choices = choices
        self.answer = answer  # JSON에서는 0~3 인덱스
        self.hint = hint
    
    def show(self):
        """문제와 선택지를 출력"""
        print(f"\n{self.question}")
        for i, choice in enumerate(self.choices, 1):  # 화면에는 1~4로 표시
            print(f"{i}) {choice}")
    
    def check_answer(self, n):
        """
        입력 번호가 정답인지 확인
        n (int): 사용자 입력 (1~4)
        return (bool): 정답 여부
        """
        return n == (self.answer + 1)  # JSON은 0~3, 사용자 입력은 1~4
    
    def show_hint(self):
        """힌트 출력"""
        if self.hint:
            print(f"💡 힌트: {self.hint}")
        else:
            print("💡 힌트가 없습니다.")
    
    def to_dict(self):
        """Quiz 객체를 dict로 변환 (JSON 저장용)"""
        return {
            "question": self.question,
            "choices": self.choices,
            "answer": self.answer,  # 0~3 인덱스로 저장
            "hint": self.hint
        }
    
    @classmethod
    def from_dict(cls, data):
        """dict에서 Quiz 객체 생성 (JSON 불러오기용)"""
        return cls(
            question=data["question"],
            choices=data["choices"],
            answer=data["answer"],  # 0~3 인덱스
            hint=data.get("hint", "")
        )

# 14. quiz_game.py 전체 틀 작성하기
# quiz_game.py

import json
from datetime import datetime
from quiz import Quiz


class QuizGame:
    def __init__(self):
        """게임 초기화"""
        # 1. 기본값 먼저 설정
        self.state_file = "state.json"
        self.quizzes = []
        self.best_score = 0
        self.best_record = {"correct": 0, "total": 0}
        self.history = []

        # 2. 저장된 데이터 불러오기
        self.load_state()

    # ──────────────────────────────────────────
    # 데이터 관련
    # ──────────────────────────────────────────

    def load_state(self):
        """state.json에서 데이터 로드"""
        try:
            with open(self.state_file, "r", encoding="utf-8") as f:
                data = json.load(f)

            self.quizzes = [Quiz.from_dict(q) for q in data.get("quizzes", [])]
            self.best_score = data.get("best_score", 0)
            self.best_record = data.get("best_record", {"correct": 0, "total": 0})
            self.history = data.get("history", [])

            print(f"✅ {len(self.quizzes)}개의 문제를 로드했습니다.")

        except FileNotFoundError:
            # 처음 실행할 때: 빈 파일 생성
            print("⚠️  처음 실행입니다. 새 파일을 생성합니다.")
            self.save_state(show_message=False)

        except json.JSONDecodeError:
            # 파일이 깨졌을 때: 기본값 유지
            print("⚠️  저장 파일이 손상되었습니다. 빈 상태로 시작합니다.")

        except Exception as e:
            # 그 외 예상치 못한 오류
            print(f"⚠️  데이터 로드 중 오류 발생: {e}")

    def save_state(self, show_message=True):
        """현재 상태를 state.json에 저장"""
        try:
            data = {
                "quizzes": [q.to_dict() for q in self.quizzes],
                "best_score": self.best_score,
                "best_record": self.best_record,
                "history": self.history
            }

            with open(self.state_file, "w", encoding="utf-8") as f:
                json.dump(data, f, indent=4, ensure_ascii=False)

            if show_message:
                print("✅ 저장되었습니다.")

        except Exception as e:
            print(f"⚠️  저장 중 오류 발생: {e}")

    # ──────────────────────────────────────────
    # 메뉴 및 메인 루프
    # ──────────────────────────────────────────

    def show_welcome(self):
        """시작 환영 메시지"""
        print("\n" + "=" * 40)
        print("   🏀 농구 퀴즈 게임에 오신 것을 환영합니다! 🏀")
        print("=" * 40)

    def show_menu(self):
        """메인 메뉴 표시"""
        print("\n" + "=" * 40)
        print("           🏀 농구 퀴즈 게임 🏀")
        print("=" * 40)
        print("  1. 게임 시작")
        print("  2. 최고 점수 보기")
        print("  3. 게임 히스토리 보기")
        print("  4. 문제 추가")
        print("  5. 문제 삭제")
        print("  6. 랜덤 게임")

    def run(self):
        """게임 메인 루프"""
        self.show_welcome()

        # 메뉴 번호 → 메서드 연결
        menu_actions = {
            "1": self.play_game,
            "2": self.show_best_score,
            "3": self.show_history,
            "4": self.add_quiz,
            "5": self.delete_quiz,
            "6": self.random_play,
            "7": self.custom_play,
        }

        while True:
            try:
                self.show_menu()
                choice = input("  선택 (0~7): ").strip()

                if choice == "0":
                    self.save_state(show_message=False)
                    print("\n👋 게임을 종료합니다. 또 만나요!\n")
                    break

                elif choice in menu_actions:
                    menu_actions[choice]()  # 해당 메서드 실행

                else:
                    print("❌ 잘못된 선택입니다. 0~7 사이의 번호를 입력하세요.")

            except KeyboardInterrupt:
                # Ctrl+C 눌렀을 때 저장 후 종료
                print("\n\n⚠️  강제 종료 감지. 저장 후 종료합니다.")
                self.save_state(show_message=False)
                break
